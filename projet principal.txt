/*
    Implementation simple timeout

    Input: count milliseconds as number

    Usage:
        setTimeout(1000) - timeout on 1 second
        setTimeout(10100) - timeout on 10 seconds and 100 milliseconds
 */
void setTimeout(int milliseconds)
{
    // If milliseconds is less or equal to 0
    // will be simple return from function without throw error
    if (milliseconds <= 0) {
        fprintf(stderr, "Count milliseconds for timeout is less or equal to 0\n");
        return;
    }

    // a current time of milliseconds
    int milliseconds_since = clock() * 1000 / CLOCKS_PER_SEC;

    // needed count milliseconds of return from this timeout
    int end = milliseconds_since + milliseconds;

    // wait while until needed time comes
    do {
        milliseconds_since = clock() * 1000 / CLOCKS_PER_SEC;
    } while (milliseconds_since <= end);
}







// Fonction pour trouver toutes les suites et renvoyer leurs coordonnées
int** findSequences(char** matrix, int h, int l, int* pointCount) {
    bool marked[h][l]; // Tableau pour marquer les positions des suites
    for (int i = 0; i < h; i++)
        for (int j = 0; j < l; j++)
            marked[i][j] = false;

    // Vérification horizontale
    for (int i = 0; i < h; i++) {
        int start = 0; // Début d'une suite
        for (int j = 1; j <= l; j++) { // On inclut un élément hors matrice pour finir les suites
            if (j == l || matrix[i][j] != matrix[i][start] && matrix[i][j]!=' ') {
                if (j - start > 3) { // Suite valide
                    for (int k = start; k < j; k++) {
                        marked[i][k] = true;
                    }
                }
                start = j;
            }
        }
    }

    // Vérification verticale
    for (int j = 0; j < l; j++) {
        int start = 0; // Début d'une suite
        for (int i = 1; i <= h; i++) { // On inclut un élément hors matrice pour finir les suites
            if (i == h || matrix[i][j] != matrix[start][j]) {
                if (i - start > 3) { // Suite valide
                    for (int k = start; k < i; k++) {
                        marked[k][j] = true;
                    }
                }
                start = i;
            }
        }
    }

    // Allocation dynamique pour le résultat
    int** result = (int**)malloc(2 * sizeof(int*));
    result[0] = (int*)malloc(MAX_POINTS * sizeof(int)); // Coordonnées x
    result[1] = (int*)malloc(MAX_POINTS * sizeof(int)); // Coordonnées y
    *pointCount = 0;

    // Collecter toutes les coordonnées marquées
    for (int i = 0; i < h; i++) {
        for (int j = 0; j < l; j++) {
            if (marked[i][j]) {
                if (*pointCount >= MAX_POINTS) break;
                result[0][*pointCount] = i; // Ligne
                result[1][*pointCount] = j; // Colonne
                (*pointCount)++;
            }
        }
    }

    return result;
}

void **afficher(int h, int l,char **matrice)
{
    printf("  ");
    for (int e=0; e<l; e++)
    {
        printf("%d ",e);
    }
    printf("\n");
    for (int i=0; i<h; i++)
    {
        printf("%d ",i);
        for (int j=0; j<l; j++)
        {
            printf("%c ",matrice[i][j]);
        }
        printf("\n");
    }
}

char **suprimer_suite(char **matrice, int **coordonnés_suite, int pointCount)
{
    int a,b;
    for (int i=0; i<pointCount; i++)
    {
        a=coordonnés_suite[0][i];
        b=coordonnés_suite[1][i];
        matrice[a][b]=' ';
    }
    return matrice;
}

char **tombé(int h, int l, char **matrice)
{
    for (int repet=0; repet<h; repet++){
        for (int i=0; i<(h-1); i++)
        {
            for (int j=0; j<l; j++)
            {
                if (matrice[i+1][j]==' ')
                {
                    matrice[i+1][j]=matrice[i][j];
                    matrice[i][j]=' ';
                }
            }
        }
    }
    return matrice;
}
